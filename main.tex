% ==================== RESEARCH ====================
\section{Research Focus and Selected Projects}
\focus{Representation learning}{Learning molecular, cellular, and patient-state representations that preserve treatment-relevant variation and generalize across biological contexts.}
\focus{Perturbation / response}{Modeling response to therapeutic and genetic interventions, including uncertainty, distributional change, and context shift.}
\focus{Structured / causal ML}{Using biological graphs, mechanistic structure, and causal assumptions for interpretable prediction and intervention reasoning.}
\vspace{1pt}
\begin{itemize}
  \item \textbf{Current representation-learning work} -- treatment-conditioned latent prediction for patient and cellular response.
  \item \textbf{DrugAgent} -- multi-agent aggregation of heterogeneous and conflicting biomedical evidence for drug discovery and repurposing.
  \item \textbf{drGT / GraphPINE} -- graph-based, interpretable drug-response modeling using drug-cell-gene structure and gene-level attribution.
\end{itemize}
